#!/bin/bash
# =============================================================================
# Hermes Agent 手动打包，在 Linux 或 WSL 的项目根目录运行：
#     bash hermes-agent打包步骤.txt
#
# 流程：检查工具 → 检查网络和令牌 → 没有内核运行时就先构建（从官方正式版里选）
#       → 校验 → 汇总检测结果，按回车后正式打包 → 列出安装包。安装包在 dist/，默认只打 x86_64。
#
# 新机器第一次运行前先装好（Debian/Ubuntu）：
#   sudo apt install -y git curl python3 tar coreutils dpkg
#   Node 26：curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/master/install.sh | bash
#            重新打开终端后运行 nvm install 26
#   uv：     curl -LsSf https://astral.sh/uv/install.sh | sh
#
# 可选环境变量：
#   PROXY=http://127.0.0.1:7890          访问 GitHub 需要代理时填写
#   GITHUB_TOKEN=令牌 或 GITHUB_TOKEN_FILE=/路径/令牌.txt
#                                        没有令牌时 GitHub 接口每小时只能调 60 次
#   HERMES_BUILD_NODE=/路径/bin/node     Node 26 不在 PATH 或 ~/.nvm 里时指定
#   HERMES_BUILD_UV=/路径/uv             uv 不在 PATH、~/.local/bin 或 ~/.cargo/bin 里时指定
#   HERMES_REF=latest 或 v0.23.1         构建内核时直接用这个版本，不再询问
#   REBUILD_CORE=1                       已有内核运行时也重新选择版本并构建
#   ALLOW_OLDER=1                        允许选比当前固定版本更旧的内核
#   PKG_VERSION=0.23.1-032               指定安装包版本号；换了内核版本时必须填写
#   FPK_ARCHES=x86_64,aarch64            同时打 ARM 包
#   SKIP_SKILL_CHECK=1                   GitHub 不可达时跳过 fnos-cli Skill 最新版核对
#   ASSUME_YES=1                         校验通过后不等按键，直接打包（没有终端时必须设）
#
# 示例：
#   PROXY=http://127.0.0.1:7890 GITHUB_TOKEN_FILE=~/github-token.txt bash hermes-agent打包步骤.txt
# =============================================================================
set -euo pipefail

info() { printf '\033[0;32m[INFO]\033[0m %s\n' "$*"; }
warn() { printf '\033[1;33m[WARN]\033[0m %s\n' "$*"; }
die()  { printf '\033[0;31m[ERROR]\033[0m %s\n' "$*" >&2; exit 1; }
step() { printf '\n\033[1;34m== %s ==\033[0m\n' "$*"; }
pin_get() { sed -n "s/^$1=//p" "$2" | head -1 | tr -d '\r'; }

cd -- "$(dirname -- "$(readlink -f -- "${BASH_SOURCE[0]}")")"
[[ "$(uname -s)" == Linux ]] || die "请在 Linux 或 WSL 中运行，不要在 Windows 下运行"
[[ -f manifest && -f build.sh && -f scripts/build-hermes.sh ]] || die "本文件要放在项目根目录，和 build.sh 同级"

# -----------------------------------------------------------------------------
step "1. 检查基础工具"
# -----------------------------------------------------------------------------
missing=()
for tool in git curl python3 tar sha256sum dpkg-deb find diff cmp stat od; do
    command -v "$tool" >/dev/null || missing+=("$tool")
done
[[ ${#missing[@]} -eq 0 ]] || die "缺少命令：${missing[*]}
       Debian/Ubuntu 可运行：sudo apt install -y git curl python3 tar coreutils diffutils findutils dpkg"
info "基础工具齐全"

# -----------------------------------------------------------------------------
step "2. 检查网络与 GitHub 令牌"
# -----------------------------------------------------------------------------
if [[ -n "${PROXY:-}" ]]; then
    export https_proxy="$PROXY" http_proxy="$PROXY" HTTPS_PROXY="$PROXY" HTTP_PROXY="$PROXY"
    info "使用代理 $PROXY"
fi
if [[ -z "${GITHUB_TOKEN:-}" && -n "${GITHUB_TOKEN_FILE:-}" ]]; then
    [[ -r "$GITHUB_TOKEN_FILE" ]] || die "读不到令牌文件 $GITHUB_TOKEN_FILE"
    GITHUB_TOKEN=$(tr -d '\r\n' < "$GITHUB_TOKEN_FILE")
    [[ -n "$GITHUB_TOKEN" ]] || die "令牌文件 $GITHUB_TOKEN_FILE 是空的"
fi
auth=()
if [[ -n "${GITHUB_TOKEN:-}" ]]; then
    export GITHUB_TOKEN
    auth=(-H "Authorization: Bearer $GITHUB_TOKEN")
fi
rate=$(curl -fsS --max-time 20 ${auth[@]+"${auth[@]}"} https://api.github.com/rate_limit |
       python3 -c 'import json,sys; r=json.load(sys.stdin)["rate"]; print(r["limit"], r["remaining"])') ||
    die "连不上 GitHub，或令牌无效。需要代理时请设置 PROXY=http://地址:端口"
read -r limit remaining <<< "$rate"
if [[ -n "${GITHUB_TOKEN:-}" ]]; then
    [[ "$limit" -ge 5000 ]] || die "令牌没有生效（GitHub 接口额度只有 $limit 次/小时），请检查令牌"
    info "令牌有效，GitHub 接口剩余 $remaining/$limit 次"
else
    warn "没有设置 GitHub 令牌，接口剩余 $remaining/$limit 次；多次运行可能被限流"
fi
[[ "$remaining" -ge 10 ]] || die "GitHub 接口剩余次数只有 $remaining 次，请稍后再试或设置令牌"

# -----------------------------------------------------------------------------
step "3. Hermes 内核运行时"
# -----------------------------------------------------------------------------
RUNTIME_PIN=runtime/hermes-version.env
if [[ ! -f "$RUNTIME_PIN" || "${REBUILD_CORE:-0}" == 1 ]]; then
    node=""
    for c in "${HERMES_BUILD_NODE:-}" "$(command -v node || true)" "$HOME"/.nvm/versions/node/v*/bin/node; do
        [[ -n "$c" && -x "$c" ]] || continue
        major=$("$c" -p 'process.versions.node.split(".")[0]' 2>/dev/null || true)
        if [[ "$major" =~ ^[0-9]+$ && "$major" -ge 26 && -x "$(dirname -- "$c")/npm" ]]; then
            node=$c; break
        fi
    done
    [[ -n "$node" ]] || die "没有找到 Node 26 或更新版本（带 npm），构建内核要用它编译网页和 TUI。
       安装：curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/master/install.sh | bash
             重新打开终端后运行 nvm install 26
       已装在别处时用 HERMES_BUILD_NODE=/路径/bin/node 指定"
    uv=""
    for c in "${HERMES_BUILD_UV:-}" "$(command -v uv || true)" "$HOME/.local/bin/uv" "$HOME/.cargo/bin/uv"; do
        [[ -n "$c" && -x "$c" ]] && { uv=$c; break; }
    done
    [[ -n "$uv" ]] || die "没有找到 uv。
       安装：curl -LsSf https://astral.sh/uv/install.sh | sh
       已装在别处时用 HERMES_BUILD_UV=/路径/uv 指定"
    export HERMES_BUILD_NODE="$node" HERMES_BUILD_UV="$uv"

    core_args=()
    [[ -n "${HERMES_REF:-}" ]] && core_args+=(--ref "$HERMES_REF")
    [[ "${REBUILD_CORE:-0}" == 1 ]] && core_args+=(--force)
    [[ "${ALLOW_OLDER:-0}" == 1 ]] && core_args+=(--allow-older)
    if [[ ! -f "$RUNTIME_PIN" ]]; then
        info "runtime/ 里还没有内核运行时，开始构建（下载源码、编译网页和 TUI，约需 5~15 分钟）"
    fi
    bash scripts/build-hermes.sh ${core_args[@]+"${core_args[@]}"}
    [[ -f "$RUNTIME_PIN" ]] || die "内核运行时没有生成，请看上面的报错"
else
    info "已有内核运行时 $(pin_get HERMES_VERSION "$RUNTIME_PIN")，跳过构建（要换版本或重建请设 REBUILD_CORE=1）"
fi

# -----------------------------------------------------------------------------
step "4. 确定安装包版本号"
# -----------------------------------------------------------------------------
core=$(pin_get HERMES_VERSION "$RUNTIME_PIN"); core=${core#v}
current=$(sed -n 's/^version[[:space:]]*=[[:space:]]*//p' manifest | head -1 | tr -d '\r"')
build_args=()
if [[ -n "${PKG_VERSION:-}" ]]; then
    [[ "$PKG_VERSION" =~ ^[0-9]+\.[0-9]+\.[0-9]+-[0-9]+$ ]] || die "PKG_VERSION 格式应为 x.y.z-序号：$PKG_VERSION"
    [[ "${PKG_VERSION%%-*}" == "$core" ]] || die "PKG_VERSION=$PKG_VERSION 与内核 $core 不一致"
    build_args=(--version "$PKG_VERSION")
    info "按指定版本号打包：$PKG_VERSION"
elif [[ "${current%%-*}" != "$core" ]]; then
    die "内核已从 ${current%%-*} 换成 $core，请指定新版本号后重新运行，例如：
       PKG_VERSION=$core-001 bash hermes-agent打包步骤.txt"
else
    info "当前版本 $current，本次打包序号自动加一"
fi

# -----------------------------------------------------------------------------
step "5. 打包前校验（不写版本、不打包）"
# -----------------------------------------------------------------------------
check_log=$(mktemp)
trap 'rm -f "$check_log"' EXIT
bash ./build.sh ${build_args[@]+"${build_args[@]}"} --check-only 2>&1 | tee "$check_log"
next_version=$(sed -n 's/.*本次版本: *//p' "$check_log" | head -1)
grep -q '仅校验模式：.*校验通过' "$check_log" || die "校验没有全部通过，请看上面的报错"
skill_line=$(sed -n 's/.*仅校验模式.*\(fnos-cli Skill [^，]*\).*/\1/p' "$check_log" | tail -1)

printf '\n\033[1;32m所有依赖检测通过：\033[0m\n'
printf '  基础工具          齐全\n'
printf '  内核运行时        %s（源码包、TUI 的 SHA256 已校验）\n' "$(pin_get HERMES_VERSION "$RUNTIME_PIN")"
printf '  私有库与许可      已校验\n'
printf '  fnos-cli Skill    %s\n' "${skill_line#fnos-cli Skill }"
printf '  本次安装包版本    %s\n' "${next_version:-未知}"
printf '  目标架构          %s\n' "${FPK_ARCHES:-x86_64}"
if [[ "${ASSUME_YES:-0}" == 1 ]]; then
    info "ASSUME_YES=1，直接开始正式打包"
elif [[ -t 0 ]]; then
    read -r -p "按回车开始正式打包，输入 q 取消：" answer
    [[ "$answer" != q && "$answer" != Q ]] || { info "已取消，没有写入版本号，也没有打包"; exit 0; }
else
    die "没有终端，无法确认。要跳过确认直接打包请设 ASSUME_YES=1"
fi

# -----------------------------------------------------------------------------
step "6. 正式打包"
# -----------------------------------------------------------------------------
bash ./build.sh ${build_args[@]+"${build_args[@]}"}

# -----------------------------------------------------------------------------
step "7. 打包结果"
# -----------------------------------------------------------------------------
version=$(sed -n 's/^version[[:space:]]*=[[:space:]]*//p' manifest | head -1 | tr -d '\r"')
shopt -s nullglob
pkgs=(dist/*_v"$version"_*.fpk)
[[ ${#pkgs[@]} -gt 0 ]] || die "dist/ 里没有找到 $version 的安装包"
ls -lh -- "${pkgs[@]}"
info "完成：版本 $version，安装包在 $(pwd)/dist/"
